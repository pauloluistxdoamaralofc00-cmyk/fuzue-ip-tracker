const { Client, GatewayIntentBits, PermissionsBitField } = require('discord.js'); // Configura o cliente do bot com as intenções necessárias const client = new Client({ intents: [ GatewayIntentBits.Guilds, GatewayIntentBits.GuildMessages, GatewayIntentBits.MessageContent, ] }); // Substitui pelo Token real do teu Bot obtido no Discord Developer Portal const TOKEN = 'O_TEU_TOKEN_DO_BOT_AQUI'; const PREFIX = '!'; // Prefixo para os comandos client.once('ready', () => { console.log(`🤖 Bot ligado com sucesso como ${client.user.tag}!`); }); client.on('messageCreate', async message => { // Ignora mensagens de outros bots ou mensagens que não começam com o prefixo if (message.author.bot || !message.content.startsWith(PREFIX)) return; const args = message.content.slice(PREFIX.length).trim().split(/ +/); const command = args.shift().toLowerCase(); // Comando de clonagem: !clone if (command === 'clone') { const sourceGuildId = args[0]; if (!sourceGuildId) { return message.reply('❌ Utilização correta: `!clone `'); } // Verifica se o bot está no servidor de destino onde o comando foi enviado const destinationGuild = message.guild; if (!destinationGuild.members.me.permissions.has(PermissionsBitField.Flags.Administrator)) { return message.reply('❌ Eu preciso de ter permissão de **Administrador** neste servidor para criar os canais e cargos!'); } try { message.reply('⏳ A iniciar o processo de clonagem. A aceder ao servidor de origem...'); // Procura o servidor de origem onde o bot também tem de estar presente const sourceGuild = client.guilds.cache.get(sourceGuildId); if (!sourceGuild) { return message.reply('❌ Não consegui encontrar o servidor de origem. Certifica-te de que o bot foi adicionado a ele!'); } // 1. Clonar Cargos (começa pelos menos importantes para manter a hierarquia ou cria do zero) message.channel.send('⚙️ A clonar cargos...'); const rolesMap = new Map(); // Para mapear cargos antigos com os novos const sortedRoles = sourceGuild.roles.cache.sort((a, b) => b.position - a.position); for (const [id, role] of sortedRoles) { if (role.name === '@everyone') continue; // Ignora o cargo padrão everyone try { const newRole = await destinationGuild.roles.create({ name: role.name, color: role.color, hoist: role.hoist, permissions: role.permissions, mentionable: role.mentionable, reason: `Clonado de ${sourceGuild.name}` }); rolesMap.set(id, newRole.id); } catch (err) { console.log(`Erro ao criar cargo ${role.name}:`, err.message); } } // 2. Clonar Categorias e Canais (Texto e Voz) message.channel.send('📁 A clonar canais e calls...'); // Ordena os canais para respeitar as categorias const channels = sourceGuild.channels.cache.sort((a, b) => a.rawPosition - b.rawPosition); for (const [id, channel] of channels) { try { if (channel.type === 4) { // Categoria (GuildCategory) await destinationGuild.channels.create({ name: channel.name, type: 4, // Categoria reason: 'Clonagem de estrutura' }); } else if (channel.type === 0 || channel.type === 2) { // Canal de Texto (0) ou Voz/Call (2) await destinationGuild.channels.create({ name: channel.name, type: channel.type, parent: channel.parentId ? destinationGuild.channels.cache.find(c => c.name === sourceGuild.channels.cache.get(channel.parentId)?.name && c.type === 4) : null, topic: channel.topic || null, bitrate: channel.bitrate || undefined, userLimit: channel.userLimit || undefined, reason: 'Clonagem de estrutura' }); } } catch (err) { console.log(`Erro ao criar canal ${channel.name}:`, err.message); } } message.channel.send(`🎉 **Clonagem concluída com sucesso!** O servidor ${destinationGuild.name} foi estruturado com base em ${sourceGuild.name}.`); } catch (error) { console.error(error); message.reply('❌ Ocorreu um erro crítico durante a clonagem. Verifica os logs da consola.'); } } }); // Inicia o bot com o Token client.login(TOKEN);
