const express = require('express'); const { createProxyMiddleware } = require('http-proxy-middleware'); const cors = require('cors'); const app = express(); const PORTA = 3000; app.use(cors()); // Registo de middleware para monitorizar e capturar informações de quem passa pelo proxy app.use((req, res, next) => { const ipCliente = req.headers['x-forwarded-for'] || req.socket.remoteAddress; console.log(`[PROXY] Pedido intercetado de IP: ${ipCliente} | Caminho: ${req.url}`); next(); }); // Configuração do Proxy para redirecionar o tráfego para um destino seguro (ex: uma API ou outro servidor) app.use('/api', createProxyMiddleware({ target: 'https://api.exemplo.com', // Substitui pelo site ou API de destino real changeOrigin: true, pathRewrite: { '^/api': '', // Remove o prefixo /api ao reencaminhar }, })); app.listen(PORTA, () => { console.log(`Servidor Proxy a funcionar na porta ${PORTA}`); });
