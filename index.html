const express = require('express'); const axios = require('axios'); const cors = require('cors'); const app = express(); app.use(cors()); app.use(express.json()); // Rota de consulta real integrada com uma API comercial de CPFs app.get('/api/consultar', async (req, res) => { const termo = req.query.termo; // Pode ser o CPF (apenas números) ou nome try { // Exemplo utilizando a estrutura de uma API real de mercado (ex: CPFHub / CPF Nacional) const apiResponse = await axios.get(`https://api.cpfhub.io/cpf/${encodeURIComponent(termo)}`, { headers: { 'x-api-key': 'SUA_API_KEY_AQUI' // Insere aqui a chave real obtida no painel do fornecedor } }); // Retorna os dados oficiais recolhidos pela API para o front-end res.json({ sucesso: true, nome: apiResponse.data.name || apiResponse.data.nome, cpf: apiResponse.data.cpf, nascimento: apiResponse.data.birthdate || apiResponse.data.nascimento, status: "REGULAR" }); } catch (error) { // Se a API externa falhar ou o CPF não for encontrado res.status(404).json({ sucesso: false, error: "Registo não encontrado ou chave de API inválida." }); } }); app.listen(3000, () => console.log('Servidor de consultas a correr na porta 3000'));
