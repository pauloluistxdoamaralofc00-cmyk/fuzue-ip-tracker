ROTAS DO PAINEL (Configurações) // ========================================== app.get('/api/configuracoes', (req, res) => { res.json(bancoDeDados); }); app.post('/api/configuracoes', (req, res) => { const { profile_image, mediafire_link } = req.body; bancoDeDados.profile_image = profile_image || null; bancoDeDados.mediafire_link = mediafire_link ? mediafire_link.trim() : null; return res.status(200).json({ sucesso: true, mensagem: "Atualizado com sucesso." }); }); app.delete('/api/configuracoes/mediafire', (req, res) => { try { bancoDeDados.mediafire_link = null; return res.status(200).json({ sucesso: true, mensagem: "Link do Mediafire limpo com sucesso." }); } catch (erro) { console.error(erro); return res.status(500).json({ sucesso: false, mensagem: "Erro interno no servidor." }); } }); // ========================================== // 2. ROTA DE PAGAMENTO (InfinitePay) // ========================================== app.post('/api/criar-pagamento', async (req, res) => { try { const { descricaoProduto, quantidade, precoEmCentavos } = req.body; const payloadInfinitePay = { handle: "djene-teixeira", items: [ { description: descricaoProduto || "Conta Free Fire / Item", quantity: quantidade || 1, price: precoEmCentavos || 2300 } ] }; const resposta = await axios.post('https://api.checkout.infinitepay.io/links', payloadInfinitePay, { headers: { 'Content-Type': 'application/json' } }); return res.status(200).json({ sucesso: true, link: resposta.data.url }); } catch (erro) { console.error("Erro ao gerar link InfinitePay:", erro.response?.data || erro.message); return res.status(500).json({ sucesso: false, mensagem: "Erro ao gerar link de pagamento." }); } }); // ========================================== // 3. Pedido: ${idPedido}`); } return.status(200).send({ recebido: true }); } catch (erro) { console.error("Erro no webhook:", erro); return.status(500).send({ erro: "Falha interna" }); } }); // ========================================== // 4.
